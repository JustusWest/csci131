# ---------------------------------------------------------------------
# 1. R4. The hours program — Example
# ---------------------------------------------------------------------

total = int(input('Enter minutes: '))

hours = total // 60
total = total % 60

print('Hours:', hours)
print('Minutes:', total)

# Output:
# Enter minutes: 145
# Hours: 2
# Minutes: 25
