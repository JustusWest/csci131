print('Done.')

# Done.


# ---------------------------------------------------------------------
# 2. else — digits.py
# ---------------------------------------------------------------------

# digits.py -- CSCI 131, Wednesday Sept 16

num = int(input('Num: '))

if num >= 10:
    print('Double digit')
else:
    print('Single digit')

print('Done.')

# Output:
# Num: 15
# Double digit
# Done.


# ---------------------------------------------------------------------
# 3. Even or odd — evenodd.py
# ---------------------------------------------------------------------

# evenodd.py

num = int(input('Enter a number: '))

if num % 2 == 0:
    print(num, 'is even')
else:
    print(num, 'is odd')

# Output:
# Enter a number: 18
# 18 is even

# ---------------------------------------------------------------------
# 7. elif — three outcomes
# ---------------------------------------------------------------------

num = int(input('Num: '))

if num > 0:
    print('Positive')
else:
    print('Negative')

# Output:
# Num: 0
# Negative


# ---------------------------------------------------------------------
# 8. elif — sign.py
# ---------------------------------------------------------------------

# sign.py

num = int(input('Num: '))

if num > 0:
    print('Positive')
elif num < 0:
    print('Negative')
else:
    print('Zero')

# Output:
# Num: 7
# Positive


# ---------------------------------------------------------------------
# 9. elif — weekday.py
# ---------------------------------------------------------------------

# weekday.py

day = int(input('Day number (1-5): '))

if day == 1:
    print('Monday')
elif day == 2:
    print('Tuesday')
elif day == 3:
    print('Wednesday')
elif day == 4:
    print('Thursday')
elif day == 5:
    print('Friday')
else:
    print('Not a weekday')

# Output:
# Day number (1-5): 4
# Thursday
