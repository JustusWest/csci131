# ---------------------------------------------------------------------
# 1. Review — an if runs its block once
# ---------------------------------------------------------------------

n = 3

if n > 0:
    print(n)
    n = n - 1

print('Done.')

# Output:
# 3
# Done.


# ---------------------------------------------------------------------
# 2. while — an if that repeats — the same program with while
# ---------------------------------------------------------------------

n = 3

while n > 0:
    print(n)
    n = n - 1

print('Done.')

# Output:
# 3
# 2
# 1
# Done.

# ---------------------------------------------------------------------
# 4. It never stops — the infinite loop
# ---------------------------------------------------------------------

n = int(input('Count down from: '))

while n > 0:
    print(n)

print('Blastoff!')

# Output:
# Count down from: 3
# 3
# 3
# 3
# 3
# 3
# ...


# ---------------------------------------------------------------------
# 5. It runs once too many (or too few) — >= instead of >
# ---------------------------------------------------------------------

n = int(input('Count down from: '))

while n >= 0:
    print(n)
    n = n - 1

print('Done')

# Output:
# Count down from: 3
# 3
# 2
# 1
# 0
# Blastoff!

# ---------------------------------------------------------------------
# 10. Putting it together — total.py — stage 2: keep a running total
# ---------------------------------------------------------------------

total = 0
line = input('Number (or done): ')

while line != 'done':
    total = total + float(line)
    line = input('Number (or done): ')

print('Total:', total)

# Output:
# Number (or done): 4
# Number (or done): 10
# Number (or done): 7
# Number (or done): done
# Total: 21.0


# ---------------------------------------------------------------------
# 11. Putting it together — total.py — stage 3: count them and average
# ---------------------------------------------------------------------

total = 0
count = 0
line = input('Number (or done): ')

while line != 'done':
    total = total + float(line)
    count = count + 1
    line = input('Number (or done): ')

print('Count:', count)
print('Total:', total)
print('Average:', total / count)

# Output:
# Number (or done): 4
# Number (or done): 10
# Number (or done): 7
# Number (or done): done
# Count: 3
# Total: 21.0
# Average: 7.0


# ---------------------------------------------------------------------
# 12. Putting it together — total.py — total.py, finished
# ---------------------------------------------------------------------

# total.py -- CSCI 131, Monday Sept 28

total = 0
count = 0
line = input('Number (or done): ')

while line != 'done':
    total = total + float(line)
    count = count + 1
    line = input('Number (or done): ')

print('Count:', count)
print('Total:', total)

if count > 0:
    print('Average:', total / count)
else:
    print('No numbers entered.')

# Output:
# Number (or done): 4
# Number (or done): 10
# Number (or done): 7
# Number (or done): done
# Count: 3
# Total: 21.0
# Average: 7.0
