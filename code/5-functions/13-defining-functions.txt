# ---------------------------------------------------------------------
# 5. Parameters — banner.py
# ---------------------------------------------------------------------

# banner.py -- CSCI 131, Monday Oct 26

def banner(word):
    line = '=' * len(word)
    print(line)
    print(word)
    print(line)

banner('Widener')
banner('CSCI 131')

# Output:
# =======
# Widener
# =======
# ========
# CSCI 131
# ========


# ---------------------------------------------------------------------
# 6. The argument's name doesn't matter — variables as arguments
# ---------------------------------------------------------------------

school = 'Widener'
banner(school)
banner(school + ' Pride')

# Output:
# =======
# Widener
# =======
# =============
# Widener Pride
# =============

# 7. Exercise solution (box.py)
# ---------------------------------------------------------------------

# box.py -- CSCI 131, Monday Oct 26

def box(word):
    edge = '*' * (len(word) + 4)
    print(edge)
    print('* ' + word + ' *')
    print(edge)

box('Widener')
box('Python')

name = input('Your name: ')
box(name)
