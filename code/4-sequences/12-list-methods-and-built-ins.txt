# ---------------------------------------------------------------------
# 1. sum, max and min — above.py with the built-ins
# ---------------------------------------------------------------------

# above.py -- CSCI 131, Wednesday Oct 21

scores = []
line = input('Score (or done): ')

while line != 'done':
    scores.append(float(line))
    line = input('Score (or done): ')

if len(scores) > 0:
    average = sum(scores) / len(scores)
    print('Average:', average)
    print('Highest:', max(scores))
    print('Lowest:', min(scores))
    print('Above average:')
    for score in scores:
        if score > average:
            print(score)
else:
    print('No scores entered.')

# Output:
# Score (or done): 70
# Score (or done): 92
# Score (or done): 85
# Score (or done): 64
# Score (or done): 79
# Score (or done): done
# Average: 78.0
# Highest: 92.0
# Lowest: 64.0
# Above average:
# 92.0
# 85.0
# 79.0


# ---------------------------------------------------------------------
# 2. Splitting on something else — total.py
# ---------------------------------------------------------------------

# total.py -- CSCI 131, Wednesday Oct 21

line = input('Scores, with commas: ')
parts = line.split(',')

scores = []
for part in parts:
    scores.append(int(part))

print(scores)
print('Average:', sum(scores) / len(scores))

# Output:
# Scores, with commas: 88,92,75
# [88, 92, 75]
# Average: 85.0
